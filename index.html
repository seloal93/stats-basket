function renderPlayerCharts(p) { if (radarChartInstance) radarChartInstance.destroy(); if (stackedChartInstance) stackedChartInstance.destroy(); const avgPts = p.pts / p.gp; const avgReb = p.rbt / p.gp; const avgAst = p.ast / p.gp; const avgStl = p.stl / p.gp; const avgBlk = p.blk / p.gp; // 1. Radar Chart con Tooltips desplazados fuera del centro const ctxRadar = document.getElementById('radarChart').getContext('2d'); radarChartInstance = new Chart(ctxRadar, { type: 'radar', data: { labels: ['PTS', 'REB', 'AST', 'STL', 'BLK'], datasets: [ { label: p.player, data: [avgPts, avgReb, avgAst, avgStl, avgBlk], backgroundColor: 'rgba(245, 158, 11, 0.35)', borderColor: '#f59e0b', borderWidth: 2, pointBackgroundColor: '#f59e0b', pointRadius: 4, pointHoverRadius: 6 }, { label: 'Media Liga', data: [ dynamicGroupAverages.pts, dynamicGroupAverages.rbt, dynamicGroupAverages.ast, dynamicGroupAverages.stl, dynamicGroupAverages.blk ], backgroundColor: 'rgba(148, 163, 184, 0.15)', borderColor: '#64748b', borderWidth: 1.5, borderDash: [4, 4], pointBackgroundColor: '#64748b', pointRadius: 3, pointHoverRadius: 5 } ] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: true, position: 'top', labels: { color: '#cbd5e1', font: { size: 10 }, boxWidth: 10 } }, tooltip: { mode: 'index', intersect: false, position: 'nearest', // Desplaza el tooltip hacia el punto activo para no tapar el centro caretPadding: 12, // Espacio entre la flecha del cuadro y el punto del radar backgroundColor: '#0f172a', titleColor: '#f59e0b', bodyColor: '#f8fafc', borderColor: '#334155', borderWidth: 1, padding: 10, callbacks: { title: function(items) { return `Categoría: ${items[0].label}`; }, label: function(ctx) { return ` ${ctx.dataset.label}: ${ctx.raw.toFixed(1)} / partido`; } } } }, scales: { r: { angleLines: { color: 'rgba(255, 255, 255, 0.1)' }, grid: { color: 'rgba(255, 255, 255, 0.1)' }, pointLabels: { color: '#94a3b8', font: { size: 10, weight: 'bold' } }, ticks: { display: false, beginAtZero: true } } } } }); // 2. Porcentajes de Producción Individual const totalSum = p.pts + p.rbt + p.ast + p.stl + p.blk; let pctPts = 0, pctReb = 0, pctAst = 0, pctStl = 0, pctBlk = 0; if (totalSum > 0) { pctPts = (p.pts / totalSum) * 100; pctReb = (p.rbt / totalSum) * 100; pctAst = (p.ast / totalSum) * 100; pctStl = (p.stl / totalSum) * 100; pctBlk = (p.blk / totalSum) * 100; } document.getElementById('stacked-desc').innerText = `Puntos (${pctPts.toFixed(0)}%) | Rebotes (${pctReb.toFixed(0)}%) | Asistencias (${pctAst.toFixed(0)}%) | Otros (${(pctStl + pctBlk).toFixed(0)}%)`; const ctxStacked = document.getElementById('stackedBarChart').getContext('2d'); stackedChartInstance = new Chart(ctxStacked, { type: 'bar', data: { labels: ['Aportación %'], datasets: [ { label: 'PTS %', data: [pctPts], backgroundColor: '#f59e0b' }, { label: 'REB %', data: [pctReb], backgroundColor: '#3b82f6' }, { label: 'AST %', data: [pctAst], backgroundColor: '#10b981' }, { label: 'STL %', data: [pctStl], backgroundColor: '#8b5cf6' }, { label: 'BLK %', data: [pctBlk], backgroundColor: '#ec4899' } ] }, options: { indexAxis: 'y', responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom', labels: { color: '#94a3b8', boxWidth: 10, font: { size: 10 } } }, tooltip: { backgroundColor: '#0f172a', borderColor: '#334155', borderWidth: 1, callbacks: { label: function(context) { return `${context.dataset.label}: ${context.raw.toFixed(1)}%`; } } } }, scales: { x: { stacked: true, max: 100, grid: { color: 'rgba(255, 255, 255, 0.05)' }, ticks: { color: '#64748b', callback: v => v + '%' } }, y: { stacked: true, grid: { display: false }, ticks: { display: false } } } } }); }
